cmake_minimum_required(VERSION 3.20)
project(disservin_evaluation LANGUAGES CXX)
add_executable(disservin_tests tests.cpp)
target_compile_features(disservin_tests PRIVATE cxx_std_17)
target_include_directories(disservin_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/build/upstream")
if(MSVC)
    target_compile_options(disservin_tests PRIVATE /W4 /EHsc)
endif()
enable_testing()
add_test(NAME disservin_minimum_contract COMMAND disservin_tests)
