cmake_minimum_required(VERSION 3.20)
project(htcw_evaluation LANGUAGES C CXX)
enable_testing()
add_library(htcw_chess STATIC vendor/htcw_chess/src/source/chess.c)
target_include_directories(htcw_chess PUBLIC vendor/htcw_chess/src)
set_target_properties(htcw_chess PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
add_executable(htcw_tests tests/test_htcw.cpp)
target_link_libraries(htcw_tests PRIVATE htcw_chess)
target_compile_features(htcw_tests PRIVATE cxx_std_14)
if(MSVC)
  target_compile_definitions(htcw_chess PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(htcw_chess PRIVATE /W4)
  target_compile_options(htcw_tests PRIVATE /W4 /WX)
endif()
foreach(case initial_moves basic_and_rejection capture en_passant en_passant_expiry castling castling_through_check pinned_piece promotions checkmate stalemate)
  add_test(NAME ${case} COMMAND htcw_tests ${case})
endforeach()
