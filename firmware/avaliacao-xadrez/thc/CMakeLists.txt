cmake_minimum_required(VERSION 3.20)
project(THCEvaluation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(thc_upstream STATIC vendor/src/thc.cpp)
target_include_directories(thc_upstream PUBLIC vendor/src)
add_executable(thc_evaluation tests/native_main.cpp tests/essential_tests.cpp)
target_link_libraries(thc_evaluation PRIVATE thc_upstream)
if(MSVC)
  target_compile_options(thc_evaluation PRIVATE /W4 /WX)
endif()
enable_testing()
foreach(case internals initial_coordinates illegal_preserves capture castling
    castling_constraints en_passant promotions san_disambiguation checkmate
    stalemate draw_queries copy_undo perft_initial perft_kiwipete)
  add_test(NAME ${case} COMMAND thc_evaluation ${case})
endforeach()
