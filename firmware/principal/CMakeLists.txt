cmake_minimum_required(VERSION 3.20)
project(FirmwarePrincipal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_subdirectory(../registro registro)
add_library(serial_game STATIC src/SerialGame.cpp)
target_include_directories(serial_game PUBLIC src)
target_link_libraries(serial_game PUBLIC game_record)
add_executable(principal_serial tests/native_serial.cpp)
target_link_libraries(principal_serial PRIVATE serial_game)
foreach(target serial_game principal_serial)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(suite lifecycle games captures castling en_passant promotions recovery protocol capacity history)
  add_test(NAME principal.${suite} COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py --exe $<TARGET_FILE:principal_serial> --suite ${suite})
endforeach()
